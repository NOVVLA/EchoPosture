# Professional Beta GPU runtime for the Debug UI and production tray.
# Standard mode still forces device="cpu"; the CUDA wheel is a superset and
# leaves StandardPoseBackend behaviour unchanged (cold import grows from ~1.6 s
# to ~4.5 s; CPU inference is unaffected).
#
# RTX 5070 Ti Laptop is Blackwell (sm_120). cu130 is used because it is the only
# index carrying the same torch/torchvision versions as the CPU runtime, so the
# switch changes the build only, never the version. cu128 would downgrade torch
# to 2.11.0 and take standard mode down with it.
#
# The official cu130 index measured ~24 KB/s here, which is unusable for a 3 GB
# wheel; the Aliyun mirror (~370 KB/s) was used instead:
#   uv pip install --python runtime\python311\python.exe \
#     --index-url https://mirrors.aliyun.com/pytorch-wheels/cu130 \
#     --extra-index-url http://pypi.tuna.tsinghua.edu.cn/simple \
#     --trusted-host pypi.tuna.tsinghua.edu.cn --index-strategy unsafe-best-match \
#     --no-python-downloads --reinstall-package torch --reinstall-package torchvision \
#     torch==2.13.0 torchvision==0.28.0
# On a fast link, --torch-backend cu130 against the official index works too.
#
# Resolve and record the exact wheel used by each packaged runtime before release.
ultralytics==8.4.120
torch==2.13.0
torchvision==0.28.0
